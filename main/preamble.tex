%% The class already provides booktabs, longtable, array, graphicx, enumitem,
%% microtype, babel, csquotes, url, hyperref, appendix, import and wallpaper.
%% Only what the content needs on top of that is loaded here.

% ----- Mathematics ----------------------------------------------------------
% amssymb is deliberately not loaded: the class pulls in newtxmath, which
% already provides \mathbb and \mathcal and clashes with amssymb.
\usepackage{amsmath}
% newtxmath (loaded by the class, hence before amsthm) already defines \openbox;
% let amsthm redefine it instead of erroring out.
\let\openbox\relax
\usepackage{amsthm}
\usepackage{mathtools}

% ----- Network names --------------------------------------------------------
% The EfficientZero networks follow MuZero's notation --- h (representation),
% g (dynamics), p (policy), v (value), vp (value prefix) --- with the parameter
% vector as an explicit argument, v(s;\theta), rather than a subscript. The
% multi-letter names need \mathit: plain math mode would set them as a product
% of single-letter variables, with the wrong spacing.
\newcommand{\vpnet}{\mathit{vp}}
\newcommand{\projnet}{\mathit{proj}}
\newcommand{\prednet}{\mathit{pred}}

% ----- Figures --------------------------------------------------------------
\graphicspath{{figures/}}
\usepackage{xcolor}
\usepackage{subcaption}
% subcaption loads caption, which disables the class's KOMA caption fonts.
% Restore them so captions keep the template's appearance (sdqthesis.cls sets
% \setkomafont{caption}{\small}, \setkomafont{captionlabel}{\sffamily\bfseries\small}
% and \setcapindent{0em}).
\captionsetup{format=plain, font=small, labelfont={sf,bf,small}}
\usepackage{float}

% ----- Diagrams -------------------------------------------------------------
\usepackage{tikz}
