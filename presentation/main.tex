\begin{frame}{Research questions}
  \begin{enumerate}

      useful policy from coverage-shaped rewards?
    \bigskip
    \item[\textbf{RQ2}] \textbf{Action space} --- program \emph{inputs} as
      actions, or \emph{mutations on a corpus}?
    \bigskip
    \item[\textbf{RQ3}] \textbf{Planning} --- does planning with the learned
      model actually help?
  \end{enumerate}
  \note{Script: NOTES.md. These three structure the whole talk; the
    conclusion answers them in order. Elaborations if asked: rewards are
    sparse and observations alias program states (RQ1); imagining how the
    target reacts (RQ3).}
\end{frame}

\begin{frame}{EfficientZero (2): planning using the learned model}
  \begin{columns}[T]
    \begin{column}{0.46\textwidth}
      \centering
      \begin{tikzpicture}[font=\scriptsize, >={Stealth[round]},
          lat/.style={draw, circle, fill=KITblue15, minimum size=6.5mm,
            inner sep=0pt}]
        \node[lat] (r) {$s^0$};
        \node[draw, rounded corners=2pt, fill=KITblack15, inner sep=3pt,
          left=9mm of r] (obs) {$o_t$};
        \draw[->] (obs) -- (r) node[midway, above, font=\tiny] {$h$};
        \node[above=1.5mm of r, text=KITblack70]
          {root: priors $p$ + Dirichlet noise};
        \node[lat, below left=6mm and 12mm of r] (c1) {};
        \node[lat, below=6mm of r] (c2) {$s^1$};
        \node[lat, below right=6mm and 12mm of r] (c3) {};
        \draw[->, KITblack50] (r) -- (c1)
          node[midway, above left, font=\tiny] {$a^1$};
        \draw[->, thick, KITblue] (r) -- (c2)
          node[midway, right=0.5mm, font=\tiny] {$a^2$};
        \draw[->, KITblack50] (r) -- (c3)
          node[midway, above right, font=\tiny] {$a^3$};
        \node[lat, below=6mm of c2, visible on=<2->] (d) {$s^2$};
        \draw[->, thick, KITblue, visible on=<2->] (c2) -- (d)
          node[midway, right=0.5mm, font=\tiny, visible on=<2->]
            {$g(s^1, a)$};
        \node[below=2mm of d, text=KITgreen, visible on=<3->]
          {evaluate: $v(s^2)$};
        \draw[->, dashed, KITred, visible on=<4->] (d) to[bend left=45]
          node[midway, left=0.5mm, font=\tiny, text=KITred,
            visible on=<4->] {back up} (c2);
        \draw[->, dashed, KITred, visible on=<4->] (c2) to[bend left=45] (r);
      \end{tikzpicture}
      {\scriptsize
      \[
        \mathrm{pUCT}(s,a)=
        \underbrace{Q(s,a)}_{\substack{\text{value}\\\text{estimate}}}
        +\;c\cdot
        \underbrace{P(s,a)}_{\substack{\text{prior}\\\text{from }p}}
        \cdot
        \underbrace{\frac{\sqrt{N(s)}}{1+N(s,a)}}_{\substack{\text{fades with}\\\text{visits }N(s,a)}}
      \]\par}
    \end{column}
    \begin{column}{0.52\textwidth}
      \begin{enumerate}
        \item<1-> \alert<1>{\textbf{select}} --- descend the tree by pUCT
        \item<2-> \alert<2>{\textbf{expand}} --- apply $g(s,a)$
        \item<3-> \alert<3>{\textbf{evaluate}} --- score the new node using $v$
        \item<4-> \alert<4>{\textbf{back up}} --- update value estimate
      \end{enumerate}
      \medskip
      \onslide<4->{%
      {\footnotesize repeat for $N{=}200$ simulations (baseline)}}
    \end{column}
  \end{columns}
  \medskip
  \centering
  \note{Script: NOTES.md. Four builds, one per step --- advance the slide as
    each step is narrated. All four networks are on the figure: $h$ at the
    root, $p$ in the priors, $g$ on the expand edge, $v$ at the leaf.
    Foreshadowing to exploit later: $T$ and $\alpha$ return as the winning
    ablation arms (T25/T15, ALPHA004); the 200 simulations return as the
    SIMS arms in RQ3.}
\end{frame}

\begin{frame}{Fuzzing as an MDP: observations and reward}
  \centering
  \begin{tikzpicture}[font=\scriptsize, >={Stealth[round]},
      box/.style={draw, rounded corners=2pt, align=center, inner sep=3pt,
        minimum height=8.5mm}]
    \node[box, fill=KITblack15] (hits) {basic-block\\hit counts};
    \node[box, fill=KITblue15, right=7mm of hits] (hash)
      {FNV hash into\\\num{4096} buckets};
    \node[box, fill=KITblue15, right=7mm of hash] (lvl)
      {AFL-style levels\\$\ell(n)=\min(1{+}\lfloor\log_2 n\rfloor,\ell_{\max})$};
    \node[box, fill=KITgreen15, right=7mm of lvl] (obs)
      {observation $o_t$\\(corpus: + seed bytes)};
    \draw[->] (hits) -- (hash);
    \draw[->] (hash) -- (lvl);
    \draw[->] (lvl) -- (obs);
  \end{tikzpicture}

  \bigskip
  \textbf{Reward: breadth + depth}
  \[
    r_t=\lambda\Bigl[\underbrace{\textstyle\sum_i\bigl(\ell(c_{t,i})-\ell(u_{t,i})\bigr)_+}_{\text{breadth: novelty vs.\ union } u_t}
    +\;\omega\underbrace{\bigl(\sigma(c_t)-d_t\bigr)_+}_{\text{depth: beat record } d_t}\Bigr],
    \qquad \sigma(c)=\textstyle\sum_i \ell(c_i)
  \]

  \medskip
  {\footnotesize reward only \emph{new} coverage to force exploration\\
   episodic: the emulator snapshot is restored at episode end}
  \note{Script: NOTES.md. Marginal-only reward is time-dependent (late
    novelty pays less); the depth-record term compensates. $\omega=0$ is the
    W0 ablation arm later.}
\end{frame}

\begin{frame}{Conclusions}
  \begin{enumerate}
    \item[\textbf{RQ1}] \textbf{Yes} --- useful policies on all six targets,
      up to an OPC UA session and invented JSON/XML nesting.
    \item[\textbf{RQ2}] \textbf{Both work; corpus-level is more flexible}
      --- classical CGF without hand-crafted operators.
    \item[\textbf{RQ3}] \textbf{Planning helps early, fades late} ---
      simulation count barely moves final coverage at this scale.
  \end{enumerate}
  \medskip
  \begin{alertblock}{Takeaway}
    Model-based deep RL can drive a fuzzer end-to-end. The main obstacle is
    execution throughput.
  \end{alertblock}
  \medskip
  \textbf{Future work}: full seed corpus (action-space scaling via
  Sampled/Gumbel MuZero, EfficientZero~V2) \textperiodcentered{}
  hybrid CGF + RL loop \textperiodcentered{} larger targets \textperiodcentered{}
  better implementation
  \note{End on the reframing: this thesis moves the question from `can RL
    fuzz?' to `can it become cheap enough?'.}
\end{frame}
